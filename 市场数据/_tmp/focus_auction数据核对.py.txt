85:         for anchor in ('MACHSCORE', 'MACHPOOL', 'MACHSIG'):
86:             a = h.count('<!--%s-->' % anchor); b = h.count('<!--/%s-->' % anchor)

88:         for anchor in ('SCORECARD', 'POOLLEDGER'):
89:             a = h.count('<!--%s-->' % anchor); b = h.count('<!--/%s-->' % anchor)
